cmake_minimum_required(VERSION 3.24)
project(shadowbane_steam_wasd VERSION 0.1.2 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Steam WASD requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /WX /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT)
enable_testing()
add_executable(steam_wasd_controls_test controls_test.cpp)
add_test(NAME steam_wasd_controls COMMAND steam_wasd_controls_test)
add_executable(steam_wasd_image_test image_test.cpp)
target_link_libraries(steam_wasd_image_test PRIVATE bcrypt)
set(STEAM_CLIENT "" CACHE FILEPATH "Official Steam client for private exact-image tests")
if(STEAM_CLIENT)
  add_test(NAME steam_wasd_image COMMAND steam_wasd_image_test "${STEAM_CLIENT}")
endif()
add_library(steam_wasd SHARED runtime.cpp)
target_link_libraries(steam_wasd PRIVATE bcrypt comctl32)
add_executable(steam_wasd_launcher launcher.cpp)
target_link_libraries(steam_wasd_launcher PRIVATE bcrypt shell32)
add_executable(steam_wasd_camera_test camera_test.cpp)
add_test(NAME steam_wasd_camera COMMAND steam_wasd_camera_test)
add_executable(steam_wasd_parent_test parent_frame_test.cpp)
add_test(NAME steam_wasd_parent COMMAND steam_wasd_parent_test)
